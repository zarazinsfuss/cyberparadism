[[extra.terms]]
name = "Ambient interface"
definition = "Information shown in the environment rather than on a device: a label floating beside a plant, a capacity reading above a walkway. The aesthetic's preferred way to show that technology is present."

[[extra.terms]]
name = "Atoll"
definition = "A ring-shaped island enclosing a lagoon. Used in the gallery for any ring-shaped enclave at sea."

[[extra.terms]]
name = "Breakaway civilization"
definition = "A small group living with technology a step ahead of the wider world, in a place of its own choosing. The default inhabitants of a Cyberparadist enclave."

[[extra.terms]]
name = "Closed loop"
definition = "A system in which energy, water, waste and food circulate with little loss. Rarely drawn, usually implied by the absence of smoke, pipes and refuse."

[[extra.terms]]
name = "Cyber Paradise, Cybergenesis, Neogenesis"
definition = "Alternative names for the aesthetic, used in different communities. This site uses Cyberparadism throughout."

[[extra.terms]]
name = "Cyberprep"
definition = "A neighbouring aesthetic of affluent, frictionless, clean futures. Shares Cyberparadism's materials; lacks its wilderness."

[[extra.terms]]
name = "Enclave"
definition = "The bounded human habitat at the center of most Cyberparadist images: a campus, garden, dome, crater city or terrace set inside a much larger natural world."

[[extra.terms]]
name = "Enclosure"
definition = "The compositional pattern in which the enclave's boundary is visible: a ring, a shell, a lattice, a shoreline."

[[extra.terms]]
name = "Ephemeralization"
definition = "Buckminster Fuller's term for doing more and more with less and less. The idea behind 'minimalism by sophistication'."

[[extra.terms]]
name = "Field guide"
definition = "This reference. A working description of the aesthetic, maintained in the open."

[[extra.terms]]
name = "Folly"
definition = "The vivid red-pink accent in the palette, used at most once per scene."

[[extra.terms]]
name = "Junglecore"
definition = "An aesthetic of dense, saturated tropical nature. Supplies much of Cyberparadism's environmental staging."

[[extra.terms]]
name = "Lattice"
definition = "A repeating structural grid, usually hexagonal or triangular, used for shells, glazing and bridges. The most common way to make a large enclosure look light."

[[extra.terms]]
name = "Minimalism by sophistication"
definition = "The principle that mature technology looks simpler, not more complicated: fewer visible machines, fewer seams, more integration."

[[extra.terms]]
name = "Motif"
definition = "A recurring visual element. In the gallery, pieces are filed by motif (dome, lattice, ring, waterfall) as well as by setting."

[[extra.terms]]
name = "Post-cyberpunk"
definition = "Work that keeps cyberpunk's advanced technology but drops its social decay. Cyberparadism is one of its moods."

[[extra.terms]]
name = "Provenance"
definition = "How a piece was made and by whom. Stated on every gallery page: creator, method, tool, date."

[[extra.terms]]
name = "Setting"
definition = "Where a piece takes place. The gallery uses five: coastal, highland, interior, sky and exoplanet."

[[extra.terms]]
name = "Signs"
definition = "In the manifesto, the technologies that point toward a cyberparadise: fusion energy, anti-gravity, synthetic life engineering, terraforming, full automation and open-source collaboration."

[[extra.terms]]
name = "Solarpunk"
definition = "A neighbouring aesthetic of green, communal, hand-maintained futures with an explicit politics. See the distinctions chapter."

[[extra.terms]]
name = "Spaceship Earth"
definition = "The planet understood as one vessel with finite resources and a shared crew. The enclave written large."

[[extra.terms]]
name = "Terraforming"
definition = "Deliberately modifying a planetary environment to make it habitable. One of the aesthetic's settings and one of the manifesto's Signs."
